# Python dependencies for maestro/ (Maestro generated-question checks).
#
# lxml: only needed for maestro/checks/html_well_formed.py (strict HTML syntax-error detection via
# lxml.etree.HTMLParser(recover=False)) and maestro/checks/tables_no_duplicates.py (table/row/cell
# tree traversal, reusing the same dependency).
#
# openpyxl: only needed for maestro/golden/generate_pool.py and import_pool.py (SME review-pool
# spreadsheet generation/import - locked cells, frozen header, dropdown validation).
#
# No known conflict between the two, or with requirements-deepeval.txt / requirements-ragas.txt, so
# maestro does not need its own venv.
lxml==6.1.1
openpyxl==3.1.5

# deepeval + anthropic: Tier 3 judge only (maestro/judge/). Confirmed to install and import
# cleanly alongside lxml/openpyxl in one venv - no conflict, so no separate venv is needed (the
# only known conflict in this repo is deepeval-vs-ragas, over click; maestro doesn't use ragas).
deepeval==4.1.5
anthropic==0.121.0

# pyyaml: test_judge_promptfoo_suite.py only - parses judge/promptfoo/synthetic_cases_suite.yaml to
# verify its content matches judge/fixtures/synthetic_cases.json (no live Promptfoo/Node run is
# needed to check this). Not used by any runtime code path.
pyyaml==6.0.3

# py-readability-metrics + nltk: maestro/checks/readability.py only (Flesch-Kincaid grade-level
# scoring, per docs/qa-context/TIER1_CHECKS_TASK.md's explicit direction to use a real readability
# library, not a hand-rolled formula). nltk's sentence tokenizer normally fetches its data
# (punkt_tab) over the network on first use - verified directly by triggering that failure - which
# would break offline CI; maestro/checks/data/nltk_data/ bundles the English-only data (~244KB) so
# this never happens. Confirmed no conflict with deepeval/click in this same venv (`pip check`).
py-readability-metrics==1.4.5
nltk==3.9.2

# pymarkdownlnt: maestro/checks/markdown_structural_compatibility.py only - a Python markdown
# linter (same rule set as the Node markdownlint tool named in the task doc), used to catch stray/
# leaked markdown syntax in an HTML field. Confirmed no conflict with deepeval/click in this venv.
pymarkdownlnt==0.9.33

# python-docx: maestro/golden/docx_convert.py only - reads real Editorial reference-article .docx
# files (paragraphs, styles, bold runs, tables) for docs/qa-context/EDITORIAL_ASSETS_STAGING_TASK.md.
python-docx==1.2.0
